### 1.3 操作前等待机制

**问题：** 页面未完全加载就操作元素

**改进方案：**

```python
# 在 browser.py 中添加智能等待
async def _wait_for_page_ready(self, context: BrowserContext, timeout: int = 10):
    """等待页面完全加载"""
    page = await context.get_current_page()
    try:
        # 等待网络空闲
        await page.wait_for_load_state("networkidle", timeout=timeout * 1000)
        # 等待DOM稳定
        await asyncio.sleep(0.5)
        return True
    except Exception:
        # 如果超时，至少等待基本加载
        await page.wait_for_load_state("domcontentloaded", timeout=5000)
        return False

# 在操作前调用
if action in ["click_element", "input_text"]:
    await self._wait_for_page_ready(context)
    # 执行操作...
```

**效果：**
- 自动等待页面加载完成
- 减少超时错误
- 提高操作稳定性

### 2.2 消息智能管理

**问题：** 消息限制策略可能丢失重要上下文

**改进方案：**

```python
# 在 memory.py 中添加消息重要性评分
class Memory(BaseModel):
    def _calculate_message_importance(self, message: Message) -> float:
        """计算消息重要性分数"""
        score = 0.0

        # System消息很重要
        if message.role == "system":
            score += 10.0

        # 包含tool_calls的assistant消息很重要
        if message.role == "assistant" and message.tool_calls:
            score += 5.0

        # 错误消息很重要
        if message.role == "tool" and "错误" in (message.content or ""):
            score += 3.0

        # 最近的消息更重要
        index = self.messages.index(message) if message in self.messages else 0
        score += (len(self.messages) - index) * 0.1

        return score

    def add_message(self, message: Message) -> None:
        self.messages.append(message)

        if len(self.messages) > self.max_messages:
            # 按重要性排序，保留重要的消息
            scored_messages = [
                (msg, self._calculate_message_importance(msg))
                for msg in self.messages
            ]
            scored_messages.sort(key=lambda x: x[1], reverse=True)

            # 保留最重要的消息
            keep_count = self.max_messages - 1  # 保留一个system消息
            important_msgs = [msg for msg, _ in scored_messages[:keep_count]]

            # 确保至少有一个system消息
            system_msgs = [msg for msg in self.messages if msg.role == "system"]
            if system_msgs and system_msgs[0] not in important_msgs:
                important_msgs = [system_msgs[0]] + important_msgs[:-1]

            self.messages = important_msgs
```

**效果：**
- 智能保留重要消息
- 减少重要上下文丢失
- 提高对话质量

### 2.3 日志系统

**问题：** 使用print输出，无法控制日志级别

**改进方案：**

```python
# 添加 logger.py
import logging
import sys
from pathlib import Path

def setup_logger(name: str = "fanghaomanus", level: str = "INFO"):
    """设置日志系统"""
    logger = logging.getLogger(name)
    logger.setLevel(getattr(logging, level.upper()))

    # 控制台处理器
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setLevel(logging.INFO)

    # 文件处理器
    log_file = Path(__file__).parent / "logs" / "app.log"
    log_file.parent.mkdir(exist_ok=True)
    file_handler = logging.FileHandler(log_file, encoding="utf-8")
    file_handler.setLevel(logging.DEBUG)

    # 格式化
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    )
    console_handler.setFormatter(formatter)
    file_handler.setFormatter(formatter)

    logger.addHandler(console_handler)
    logger.addHandler(file_handler)

    return logger

# 在代码中使用
from logger import setup_logger
logger = setup_logger()

logger.info("Agent开始执行")
logger.debug(f"工具调用: {tool_calls}")
logger.error(f"操作失败: {error}")
```

**效果：**
- 统一的日志管理
- 支持不同日志级别
- 便于调试和问题追踪
